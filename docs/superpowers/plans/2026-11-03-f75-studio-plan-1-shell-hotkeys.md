# F75 Studio — Plan 1: App Shell, Hotkey Builder, Script Manager

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a Windows 11 tray app (WPF + Fluent UI) with a visual AutoHotkey v2 hotkey builder, a script manager for the user's `.ahk` files, and a settings page.

**Architecture:** A .NET 8 solution with two plain libraries (`F75Studio.Core` for storage/single-instance/logging, `F75Studio.Hotkeys` for the AHK model, generator, importer and process runner) and a WPF app (`F75Studio.App`) using MVVM (CommunityToolkit.Mvvm). The hotkey cards in `hotkeys.json` are the source of truth; `hotkeys.ahk` is regenerated, validated with AutoHotkey `/validate`, then run.

**Tech Stack:** .NET 8, WPF, WPF-UI 4.3.0, H.NotifyIcon.Wpf 2.4.1, CommunityToolkit.Mvvm 8.4.2, AvalonEdit 6.3.1.120, xUnit.

This is Plan 1 of 3 for the spec `docs/superpowers/specs/2026-11-03-f75-studio-design.md`:
- **Plan 1 (this):** build-order steps 1–2 (shell, tray, settings, hotkeys, scripts).
- **Plan 2:** lighting editor + profiles against `FakeKeyboardDevice`, `--profile` handling, tray lighting submenu.
- **Plan 3:** USB capture analysis → `AulaF75Device`, device diagnostics in Settings. Written after captures exist.

**Deliberate deviations from the spec (approved scope trims):**
- Adds a small `F75Studio.Core` project (JSON store, paths, single instance, log) so both feature libraries and Plan 2 share it.
- "Start with Windows" state lives in the registry Run key, not `settings.json` (single source of truth).
- `AhkImporter` turns unparseable *hotkey definitions* into Custom AHK cards; non-hotkey top-level code (hotstrings, global code, other directives) is reported as "skipped lines" instead of imported.
- Single-instance forwarding is implemented now but only `--show` is handled; `--profile` handling is Plan 2.

```csharp
        var full = Combine(relativePath);
        Directory.CreateDirectory(System.IO.Path.GetDirectoryName(full)!);

        return full;
    }

    }
```

```csharp
using System.Text.Json;
using System.Text.Json.Serialization;

namespace F75Studio.Core;

public sealed class JsonFileStore<T> where T : class
{
    public static readonly JsonSerializerOptions Options = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        Converters = { new JsonStringEnumConverter(JsonNamingPolicy.CamelCase) },
    };

    private readonly Func<T> _createDefault;

    public JsonFileStore(string filePath, Func<T> createDefault)
    {
        FilePath = filePath;
        _createDefault = createDefault;
    }

    public string FilePath { get; }

    public T Load()
    {
        if (!File.Exists(FilePath)) return _createDefault();
        try
        {
            return JsonSerializer.Deserialize<T>(File.ReadAllText(FilePath), Options) ?? _createDefault();
        }
        catch (JsonException)
        {
            File.Copy(FilePath, FilePath + ".corrupt", overwrite: true);
            return _createDefault();
        }
    }

    public void Save(T value)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(FilePath)!);
        var temp = FilePath + ".tmp";
        File.WriteAllText(temp, JsonSerializer.Serialize(value, Options));
        File.Move(temp, FilePath, overwrite: true);
    }
}
```

```csharp
using System.Text.Json.Serialization;

namespace F75Studio.Hotkeys.Model;

public enum MediaCommand { PlayPause, Next, Previous, Stop, VolumeUp, VolumeDown, Mute }

[JsonPolymorphic(TypeDiscriminatorPropertyName = "type")]
[JsonDerivedType(typeof(SendKeysAction), "send")]
[JsonDerivedType(typeof(RemapAction), "remap")]
[JsonDerivedType(typeof(TypeTextAction), "text")]
[JsonDerivedType(typeof(RunAction), "run")]
[JsonDerivedType(typeof(MediaAction), "media")]
[JsonDerivedType(typeof(SwitchLightingProfileAction), "lighting")]
[JsonDerivedType(typeof(CustomCodeAction), "code")]
[JsonDerivedType(typeof(DisableKeyAction), "disable")]
public abstract record HotkeyAction;

/// <summary>Keys in AutoHotkey Send syntax, e.g. "^{Left}".</summary>
public sealed record SendKeysAction(string Keys, bool Blind = false) : HotkeyAction;

public sealed record RemapAction(string TargetKey) : HotkeyAction;

public sealed record TypeTextAction(string Text) : HotkeyAction;

public sealed record RunAction(string Target, string? Arguments = null) : HotkeyAction;

public sealed record MediaAction(MediaCommand Command) : HotkeyAction;

public sealed record SwitchLightingProfileAction(string ProfileName) : HotkeyAction;

/// <summary>Raw AutoHotkey v2 code placed inside the hotkey's braces.</summary>
public sealed record CustomCodeAction(string Code) : HotkeyAction;

public sealed record DisableKeyAction() : HotkeyAction;
```

```csharp
using F75Studio.Hotkeys.Model;

namespace F75Studio.Hotkeys;

public static class MediaKeys
{
    private static readonly Dictionary<MediaCommand, string> Map = new()
    {
        [MediaCommand.PlayPause] = "{Media_Play_Pause}",
        [MediaCommand.Next] = "{Media_Next}",
        [MediaCommand.Previous] = "{Media_Prev}",
        [MediaCommand.Stop] = "{Media_Stop}",
        [MediaCommand.VolumeUp] = "{Volume_Up}",
        [MediaCommand.VolumeDown] = "{Volume_Down}",
        [MediaCommand.Mute] = "{Volume_Mute}",
    };

    public static string ToSendKey(MediaCommand command) => Map[command];

    public static bool TryParse(string sendKeys, out MediaCommand command)
    {
        foreach (var (key, value) in Map)
        {
            if (string.Equals(value, sendKeys.Trim(), StringComparison.OrdinalIgnoreCase))
            {
                command = key;
                return true;
            }
        }
        command = default;
        return false;
    }
}
```

```csharp
using System.Text;

namespace F75Studio.Hotkeys;

/// <summary>AutoHotkey v2 quoted string literals (escape character is the backtick).</summary>
public static class AhkString
{
    public static string Quote(string value)
    {
        var sb = new StringBuilder(value.Length + 2).Append('"');
        foreach (var c in value)
        {
            sb.Append(c switch
            {
                '`' => "``",
                '"' => "`\"",
                ';' => "`;",
                '\n' => "`n",
                '\r' => "`r",
                '\t' => "`t",
                _ => c.ToString(),
            });
        }
        return sb.Append('"').ToString();
    }

    /// <summary>Parses a "..." or '...' literal at the start of <paramref name="text"/>.</summary>
    public static bool TryUnquote(string text, out string value, out string rest)
    {
        value = "";
        rest = text;
        if (text.Length == 0 || (text[0] != '"' && text[0] != '\'')) return false;

        var quote = text[0];
        var sb = new StringBuilder();
        for (var i = 1; i < text.Length; i++)
        {
            var c = text[i];
            if (c == '`' && i + 1 < text.Length)
            {
                var next = text[++i];
                sb.Append(next switch { 'n' => '\n', 'r' => '\r', 't' => '\t', _ => next });
                continue;
            }
            if (c == quote)
            {
                value = sb.ToString();
                rest = text[(i + 1)..];
                return true;
            }
            sb.Append(c);
        }
        return false;
    }
}
```

```csharp
using F75Studio.Hotkeys.Model;

namespace F75Studio.Hotkeys;

/// <param name="Text">Full script, CRLF line endings.</param>
/// <param name="LineOwners">1-based line number → id of the hotkey that produced it.</param>
public sealed record GeneratedScript(string Text, IReadOnlyDictionary<int, Guid> LineOwners);

public sealed class AhkGenerator
{
    public const string Header = "; Generated by F75 Studio. Do not edit: changes are overwritten when you click Apply.";

    private readonly string _appExecutablePath;

    public AhkGenerator(string appExecutablePath) => _appExecutablePath = appExecutablePath;

    public GeneratedScript Generate(IEnumerable<Hotkey> hotkeys)
    {
        var lines = new List<string> { Header, "#Requires AutoHotkey v2.0", "#SingleInstance Force" };
        var owners = new Dictionary<int, Guid>();
        var enabled = hotkeys.Where(h => h.Enabled).ToList();

        var scopes = enabled
            .Select(h => NormalizeScope(h.AppScope))
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .OrderBy(s => s is null ? 0 : 1)
            .ThenBy(s => s, StringComparer.OrdinalIgnoreCase);

        foreach (var scope in scopes)
        {
            lines.Add("");
            if (scope is not null) lines.Add($"#HotIf WinActive({AhkString.Quote("ahk_exe " + scope)})");

            string? group = null;
            foreach (var hotkey in enabled.Where(h => string.Equals(NormalizeScope(h.AppScope), scope, StringComparison.OrdinalIgnoreCase)))
            {
                if (hotkey.Group != group)
                {
                    group = hotkey.Group;
                    lines.Add($"; {group}");
                }
                foreach (var line in Render(hotkey))
                {
                    lines.Add(line);
                    owners[lines.Count] = hotkey.Id;
                }
            }

            if (scope is not null) lines.Add("#HotIf");
        }

        return new GeneratedScript(string.Join("\r\n", lines) + "\r\n", owners);
    }

    private static string? NormalizeScope(string? scope) => string.IsNullOrWhiteSpace(scope) ? null : scope.Trim();

    private IEnumerable<string> Render(Hotkey hotkey)
    {
        var t = hotkey.Trigger.Trim();
        switch (hotkey.Action)
        {
            case SendKeysAction a:
                yield return $"{t}::Send {AhkString.Quote((a.Blind ? "{Blind}" : "") + a.Keys)}";
                break;
            case RemapAction a:
                yield return $"{t}::{a.TargetKey.Trim()}";
                break;
            case TypeTextAction a:
                yield return $"{t}::SendText {AhkString.Quote(a.Text)}";
                break;
            case RunAction a:
                yield return $"{t}::Run {AhkString.Quote(RunCommand(a.Target, a.Arguments))}";
                break;
            case MediaAction a:
                yield return $"{t}::Send {AhkString.Quote(MediaKeys.ToSendKey(a.Command))}";
                break;
            case SwitchLightingProfileAction a:
                yield return $"{t}::Run {AhkString.Quote(RunCommand(_appExecutablePath, $"--profile \"{a.ProfileName}\""))}";
                break;
            case CustomCodeAction a:
                yield return $"{t}::";
                yield return "{";
                foreach (var line in a.Code.ReplaceLineEndings("\n").Split('\n'))
                    yield return line.Length == 0 ? "" : "    " + line;
                yield return "}";
                break;
            case DisableKeyAction:
                yield return $"{t}::Return";
                break;
            default:
                throw new NotSupportedException($"Unknown action {hotkey.Action.GetType().Name}");
        }
    }

    private static string RunCommand(string target, string? arguments)
    {
        var quoted = target.Contains(' ') && !target.Contains("://") ? $"\"{target}\"" : target;
        return string.IsNullOrWhiteSpace(arguments) ? quoted : $"{quoted} {arguments.Trim()}";
    }
}
```

```csharp
using System.Text.RegularExpressions;
using F75Studio.Hotkeys.Model;

namespace F75Studio.Hotkeys;

public sealed record ImportResult(IReadOnlyList<Hotkey> Hotkeys, IReadOnlyList<string> SkippedLines);

public static partial class AhkImporter
{
    private static readonly HashSet<string> IgnoredDirectives = new(StringComparer.OrdinalIgnoreCase) { "#Requires", "#SingleInstance" };

    [GeneratedRegex(@"^(?<trigger>[^:;\s""'(][^:""'(]*?)\s*::(?<rest>.*)$")]
    private static partial Regex HotkeyLine();

    [GeneratedRegex(@"^#HotIf\s+WinActive\(\s*[""']ahk_exe\s+(?<exe>[^""']+)[""']\s*\)\s*$", RegexOptions.IgnoreCase)]
    private static partial Regex HotIfExe();

    [GeneratedRegex(@"^[A-Za-z0-9_]+$|^\S$")]
    private static partial Regex KeyName();

    public static ImportResult Import(string scriptText, string defaultGroup)
    {
        var lines = scriptText.ReplaceLineEndings("\n").Split('\n');
        var hotkeys = new List<Hotkey>();
        var skipped = new List<string>();
        var group = defaultGroup;
        string? scope = null;

        for (var i = 0; i < lines.Length; i++)
        {
            var line = lines[i].Trim();
            if (line.Length == 0) continue;

            if (line.StartsWith(';'))
            {
                var text = line.TrimStart(';').Trim();
                if (text.Length > 0 && !line.Equals(AhkGenerator.Header, StringComparison.Ordinal)) group = text;
                continue;
            }

            if (line.StartsWith('#'))
            {
                if (IgnoredDirectives.Contains(line.Split(' ', 2)[0])) continue;
                if (line.Equals("#HotIf", StringComparison.OrdinalIgnoreCase)) { scope = null; continue; }
                var hotIf = HotIfExe().Match(line);
                if (hotIf.Success) { scope = hotIf.Groups["exe"].Value.Trim(); continue; }
                skipped.Add(line);
                continue;
            }

            var match = HotkeyLine().Match(line);
            if (!match.Success) { skipped.Add(line); continue; }

            var trigger = match.Groups["trigger"].Value.Trim();
            var rest = StripComment(match.Groups["rest"].Value).Trim();

            HotkeyAction action;
            if (rest.Length == 0 || rest == "{")
            {
                var body = ReadBlock(lines, ref i, openedOnSameLine: rest == "{");
                if (body is null) { skipped.Add(line); continue; }
                action = new CustomCodeAction(body);
            }
            else
            {
                action = ParseAction(rest);
            }

            hotkeys.Add(new Hotkey { Trigger = trigger, Group = group, AppScope = scope, Action = action });
        }

        return new ImportResult(hotkeys, skipped);
    }

    private static HotkeyAction ParseAction(string rest)
    {
        if (rest.Equals("Return", StringComparison.OrdinalIgnoreCase)) return new DisableKeyAction();
        if (TryCall(rest, "SendText", out var text)) return new TypeTextAction(text);
        if (TryCall(rest, "Send", out var keys))
        {
            if (keys.StartsWith("{Blind}", StringComparison.OrdinalIgnoreCase))
                return new SendKeysAction(keys["{Blind}".Length..], Blind: true);
            return MediaKeys.TryParse(keys, out var media) ? new MediaAction(media) : new SendKeysAction(keys);
        }
        if (TryCall(rest, "Run", out var command)) return ParseRun(command);
        if (KeyName().IsMatch(rest)) return new RemapAction(rest);
        return new CustomCodeAction(rest);
    }

    /// <summary>Matches <c>Name "literal"</c> or <c>Name("literal")</c> with nothing after it.</summary>
    private static bool TryCall(string rest, string name, out string argument)
    {
        argument = "";
        if (!rest.StartsWith(name, StringComparison.OrdinalIgnoreCase) || rest.Length == name.Length) return false;

        var after = rest[name.Length..];
        var paren = after[0] == '(';
        if (!paren && !char.IsWhiteSpace(after[0])) return false;

        after = (paren ? after[1..] : after).Trim();
        if (!AhkString.TryUnquote(after, out argument, out var tail)) return false;
        tail = tail.Trim();
        return paren ? tail == ")" : tail.Length == 0;
    }

    private static HotkeyAction ParseRun(string command)
    {
        string target, arguments;
        if (command.StartsWith('"'))
        {
            var end = command.IndexOf('"', 1);
            if (end < 0) return new RunAction(command);
            target = command[1..end];
            arguments = command[(end + 1)..].Trim();
        }
        else
        {
            var space = command.IndexOf(' ');
            target = space < 0 ? command : command[..space];
            arguments = space < 0 ? "" : command[(space + 1)..].Trim();
        }

        if (Path.GetFileName(target).Equals("F75Studio.exe", StringComparison.OrdinalIgnoreCase)
            && arguments.StartsWith("--profile ", StringComparison.Ordinal))
        {
            return new SwitchLightingProfileAction(arguments["--profile ".Length..].Trim().Trim('"'));
        }

        return new RunAction(target, arguments.Length == 0 ? null : arguments);
    }

    /// <summary>Reads a { ... } block after a hotkey line; returns the dedented body or null.</summary>
    private static string? ReadBlock(string[] lines, ref int index, bool openedOnSameLine)
    {
        var j = index + 1;
        if (!openedOnSameLine)
        {
            while (j < lines.Length && lines[j].Trim().Length == 0) j++;
            if (j >= lines.Length || lines[j].Trim() != "{") return null;
            j++;
        }

        var depth = 1;
        var body = new List<string>();
        for (; j < lines.Length; j++)
        {
            var raw = lines[j];
            depth += raw.Count(c => c == '{') - raw.Count(c => c == '}');
            if (depth <= 0)
            {
                index = j;
                return Dedent(body);
            }
            body.Add(raw.TrimEnd());
        }
        return null;
    }

    private static string Dedent(List<string> body)
    {
        var indent = body.Where(l => l.Trim().Length > 0)
            .Select(l => l.Length - l.TrimStart().Length)
            .DefaultIfEmpty(0)
            .Min();
        return string.Join("\n", body.Select(l => l.Length >= indent ? l[indent..] : l.TrimStart())).Trim('\n');
    }

    /// <summary>Removes a trailing " ; comment" that is not inside a string literal.</summary>
    private static string StripComment(string text)
    {
        char? quote = null;
        for (var i = 0; i < text.Length; i++)
        {
            var c = text[i];
            if (quote is not null)
            {
                if (c == '`') i++;
                else if (c == quote) quote = null;
            }
            else if (c is '"' or '\'')
            {
                quote = c;
            }
            else if (c == ';' && i > 0 && char.IsWhiteSpace(text[i - 1]))
            {
                return text[..i];
            }
        }
        return text;
    }
}
```

```csharp
using System.Text;

namespace F75Studio.Hotkeys;

/// <summary>Turns a stream of key presses into an AHK trigger. Complete once every key is released.</summary>
public sealed class TriggerRecorder
{
    private readonly List<string> _held = [];

    public string? Trigger { get; private set; }
    public bool IsComplete { get; private set; }

    public void KeyDown(string ahkKey)
    {
        if (IsComplete || _held.Contains(ahkKey)) return;
        _held.Add(ahkKey);
        if (AhkKeyNames.IsModifier(ahkKey)) return;

        var heldNormalKey = _held.FirstOrDefault(k => k != ahkKey && !AhkKeyNames.IsModifier(k));
        Trigger = heldNormalKey is not null ? $"{heldNormalKey} & {ahkKey}" : ModifierPrefix() + ahkKey;
    }

    public void KeyUp(string ahkKey)
    {
        if (IsComplete) return;
        if (Trigger is null && _held is [var only] && only == ahkKey && AhkKeyNames.IsModifier(ahkKey)) Trigger = ahkKey;
        _held.Remove(ahkKey);
        if (_held.Count == 0 && Trigger is not null) IsComplete = true;
    }

    private string ModifierPrefix()
    {
        var sb = new StringBuilder();
        if (_held.Any(k => k is "LWin" or "RWin")) sb.Append('#');
        if (_held.Any(k => k is "LCtrl" or "RCtrl")) sb.Append('^');
        if (_held.Any(k => k is "LAlt" or "RAlt")) sb.Append('!');
        if (_held.Any(k => k is "LShift" or "RShift")) sb.Append('+');
        return sb.ToString();
    }
}
```

```csharp
namespace F75Studio.Hotkeys;

public static partial class AhkErrorParser
{
    [GeneratedRegex(@"\((?<line>\d+)\)\s*:\s*==>\s*(?<msg>[^\r\n]+)")]
    private static partial Regex ErrorLine();

    [GeneratedRegex(@"Specifically:\s*(?<detail>[^\r\n]+)")]
    private static partial Regex Specifically();

    public static AhkValidationResult Parse(string output)
    {
        var match = ErrorLine().Match(output);
        if (!match.Success)
        {
            var raw = string.IsNullOrWhiteSpace(output) ? "AutoHotkey reported an unknown error." : output.Trim();
            return new AhkValidationResult(false, raw, null);
        }

        var message = match.Groups["msg"].Value.Trim();
        var detail = Specifically().Match(output);
        if (detail.Success) message += $" ({detail.Groups["detail"].Value.Trim()})";
        return new AhkValidationResult(false, message, int.Parse(match.Groups["line"].Value));
    }
}
```

```csharp
namespace F75Studio.Hotkeys;

/// <summary>Asks a running AutoHotkey process to exit like its tray "Exit" item, so it removes its tray icon.</summary>
internal static class AhkWindow
{
    private const uint WmCommand = 0x0111;
    private const nint IdTrayExit = 65307;

    public static bool RequestExit(int processId)
    {
        var posted = false;
        EnumWindows((hwnd, _) =>
        {
            GetWindowThreadProcessId(hwnd, out var pid);
            if (pid != (uint)processId) return true;
            var className = new StringBuilder(64);
            GetClassName(hwnd, className, className.Capacity);
            if (className.ToString() != "AutoHotkey") return true;
            posted = PostMessage(hwnd, WmCommand, IdTrayExit, 0);
            return false;
        }, 0);
        return posted;
    }

    [DllImport("user32.dll")]
    private static extern bool EnumWindows(EnumWindowsProc callback, nint lParam);

    [DllImport("user32.dll")]
    private static extern uint GetWindowThreadProcessId(nint hwnd, out uint processId);

    [DllImport("user32.dll", CharSet = CharSet.Unicode)]
```

```csharp
using System.Diagnostics;
using System.Text;

namespace F75Studio.Hotkeys;

public sealed class AhkRunner : IAhkRunner, IDisposable
{
    private static readonly TimeSpan ValidateTimeout = TimeSpan.FromSeconds(10);
    private readonly Dictionary<string, Process> _processes = new(StringComparer.OrdinalIgnoreCase);
    private readonly object _gate = new();

    public AhkRunner(string? ahkExePath) => AhkExePath = ahkExePath;

    public string? AhkExePath { get; set; }

    public event EventHandler<string>? RunningChanged;

    public IReadOnlyCollection<string> RunningScripts
    {
        get { lock (_gate) return _processes.Where(p => !p.Value.HasExited).Select(p => p.Key).ToList(); }
    }

    public async Task<AhkValidationResult> ValidateAsync(string scriptPath, CancellationToken ct = default)
    {
        var psi = new ProcessStartInfo(RequireExe())
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardError = true,
            RedirectStandardOutput = true,
            StandardErrorEncoding = Encoding.UTF8,
            StandardOutputEncoding = Encoding.UTF8,
        };
        psi.ArgumentList.Add("/ErrorStdOut=UTF-8");
        psi.ArgumentList.Add("/validate");
        psi.ArgumentList.Add(Path.GetFullPath(scriptPath));

        using var process = Process.Start(psi)!;
        var stderr = process.StandardError.ReadToEndAsync();
        var stdout = process.StandardOutput.ReadToEndAsync();

        using var timeout = CancellationTokenSource.CreateLinkedTokenSource(ct);
        timeout.CancelAfter(ValidateTimeout);
        try
        {
            await process.WaitForExitAsync(timeout.Token);
        }
        catch (OperationCanceledException) when (!ct.IsCancellationRequested)
        {
            process.Kill();
            return new AhkValidationResult(false, "AutoHotkey did not finish validating within 10 seconds.", null);
        }

        var output = await stderr + await stdout;
        return process.ExitCode == 0 ? AhkValidationResult.Ok : AhkErrorParser.Parse(output);
    }

    public void Start(string scriptPath)
    {
        var exe = RequireExe();
        var full = Path.GetFullPath(scriptPath);
        Stop(full);

        var psi = new ProcessStartInfo(exe) { UseShellExecute = false, WorkingDirectory = Path.GetDirectoryName(full)! };
        psi.ArgumentList.Add(full);
        var process = Process.Start(psi)!;
        process.EnableRaisingEvents = true;
        process.Exited += (_, _) =>
        {
            lock (_gate)
            {
                if (_processes.TryGetValue(full, out var current) && current == process) _processes.Remove(full);
            }
            RunningChanged?.Invoke(this, full);
        };

        lock (_gate) _processes[full] = process;
        RunningChanged?.Invoke(this, full);
    }

    public void Stop(string scriptPath)
    {
        var full = Path.GetFullPath(scriptPath);
        Process? process;
        lock (_gate)
        {
            if (!_processes.Remove(full, out process)) return;
        }

        try
        {
            if (!process.HasExited && !(AhkWindow.RequestExit(process.Id) && process.WaitForExit(1500)))
            {
                process.Kill();
                process.WaitForExit(1500);
            }
        }
        catch (InvalidOperationException)
        {
            // Process already exited.
        }
        finally
        {
            process.Dispose();
        }
        RunningChanged?.Invoke(this, full);
    }

    public bool IsRunning(string scriptPath)
    {
        lock (_gate) return _processes.TryGetValue(Path.GetFullPath(scriptPath), out var p) && !p.HasExited;
    }

    public void StopAll()
    {
        List<string> paths;
        lock (_gate) paths = _processes.Keys.ToList();
        foreach (var path in paths) Stop(path);
    }

    public void Dispose() => StopAll();

    private string RequireExe() =>
        AhkExePath is { } exe && File.Exists(exe) ? exe : throw new AhkNotFoundException();
}
```

```csharp
        _mutex = new Mutex(initiallyOwned: true, $@"Local\{name}.Mutex", out var createdNew);
        IsFirstInstance = createdNew;
        if (IsFirstInstance) _ = ListenAsync(_cts.Token);
    }

    public bool IsFirstInstance { get; }

    /// <summary>Raised on a background thread with the arguments another instance was started with.</summary>
    public event EventHandler<string[]>? ArgumentsReceived;

    public static async Task<bool> SendToFirstInstanceAsync(string name, string[] args, TimeSpan timeout)
    {
        try
        {
            await using var client = new NamedPipeClientStream(".", PipeName(name), PipeDirection.Out);
            await client.ConnectAsync((int)timeout.TotalMilliseconds);
            await using var writer = new StreamWriter(client, new UTF8Encoding(false));
            await writer.WriteAsync(JsonSerializer.Serialize(args));
            await writer.FlushAsync();
            return true;
        }
        catch (Exception ex) when (ex is TimeoutException or IOException)
        {
            return false;
        }
    }

    private async Task ListenAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            try
            {
                await using var server = new NamedPipeServerStream(PipeName(_name), PipeDirection.In, 1, PipeTransmissionMode.Byte, PipeOptions.Asynchronous);
                await server.WaitForConnectionAsync(ct);
                using var reader = new StreamReader(server, Encoding.UTF8);
                var args = JsonSerializer.Deserialize<string[]>(await reader.ReadToEndAsync(ct)) ?? [];
                ArgumentsReceived?.Invoke(this, args);
            }
            catch (OperationCanceledException)
            {
                return;
            }
            catch (Exception ex) when (ex is IOException or JsonException)
            {
                // A client sent garbage or disconnected early; keep listening.
            }
        }
    }

    private static string PipeName(string name) => $"{name}.Pipe.{Environment.UserName}";

    public void Dispose()
    {
        _cts.Cancel();
        _mutex.Dispose();
    }
}
```

```csharp
using System.Collections.ObjectModel;
using System.ComponentModel;
using System.IO;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using F75Studio.App.Services;
using F75Studio.Hotkeys;
using Wpf.Ui.Controls;

namespace F75Studio.App.ViewModels;

public sealed partial class HotkeysViewModel : ObservableObject
{
    private readonly AppServices _services;
    private readonly string _scriptPath;
    private TriggerRecorder? _recorder;
    private bool _resumeAfterRecording;

    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(DeleteCommand), nameof(ToggleRecordingCommand))]
    private HotkeyItemViewModel? _selectedItem;

    [ObservableProperty] private string _preview = "";
    [ObservableProperty] private bool _isDirty;
    [ObservableProperty] private bool _isRecording;
    [ObservableProperty] private string? _statusMessage;
    [ObservableProperty] private bool _isStatusOpen;
    [ObservableProperty] private InfoBarSeverity _statusSeverity = InfoBarSeverity.Informational;

    public HotkeysViewModel(AppServices services)
    {
        _services = services;
        _scriptPath = services.Paths.GeneratedHotkeysScript;
        foreach (var hotkey in services.HotkeyStore.Load()) AddItem(HotkeyItemViewModel.FromHotkey(hotkey));
        Items.CollectionChanged += (_, _) => { IsDirty = true; RefreshPreview(); };
        SelectedItem = Items.FirstOrDefault();
        RefreshPreview();
        IsDirty = false;
    }

    public ObservableCollection<HotkeyItemViewModel> Items { get; } = [];

    [RelayCommand]
    private void AddHotkey()
    {
        var item = new HotkeyItemViewModel { Group = SelectedItem?.Group ?? "General" };
        AddItem(item);
        SelectedItem = item;
    }

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void Delete()
    {
        var index = Items.IndexOf(SelectedItem!);
        Items.RemoveAt(index);
        SelectedItem = Items.Count == 0 ? null : Items[Math.Min(index, Items.Count - 1)];
    }

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void ToggleRecording()
    {
        if (IsRecording) StopRecording();
        else BeginRecording();
    }

    [RelayCommand]
    private async Task ApplyAsync()
    {
        foreach (var item in Items) item.Error = null;
        var missing = Items.Where(i => string.IsNullOrWhiteSpace(i.Trigger)).ToList();
        if (missing.Count > 0)
        {
            foreach (var item in missing) item.Error = "Set a trigger first.";
            SelectedItem = missing[0];
            SetStatus($"{missing.Count} hotkey(s) have no trigger.", InfoBarSeverity.Error);
            return;
        }

        var hotkeys = Items.Select(i => i.ToHotkey()).ToList();
        _services.HotkeyStore.Save(hotkeys);
        var script = _services.Generator.Generate(hotkeys);
        var pending = Path.ChangeExtension(_scriptPath, ".pending.ahk");
        ScriptFile.Write(pending, script.Text);

        try
        {
            var result = await _services.Runner.ValidateAsync(pending);
            if (!result.Success)
            {
                File.Delete(pending);
                var owner = result.Line is int line && script.LineOwners.TryGetValue(line, out var id)
                    ? Items.FirstOrDefault(i => i.Id == id)
                    : null;
                if (owner is not null)
                {
                    owner.Error = result.Message;
                    SelectedItem = owner;
                }
                SetStatus($"AutoHotkey rejected the script: {result.Message}", InfoBarSeverity.Error);
                return;
            }

            File.Move(pending, _scriptPath, overwrite: true);
            _services.Runner.Start(_scriptPath);
            IsDirty = false;
            SetStatus($"Applied {hotkeys.Count(h => h.Enabled)} hotkey(s).", InfoBarSeverity.Success);
        }
        catch (AhkNotFoundException)
        {
            File.Delete(pending);
            SetStatus("AutoHotkey v2 was not found. Set its path in Settings. Your hotkeys are saved.", InfoBarSeverity.Error);
        }
    }

    public void RecordKeyDown(string ahkKey)
    {
        if (_recorder is null) return;
        _recorder.KeyDown(ahkKey);
        ShowRecordedTrigger();
    }

    public void RecordKeyUp(string ahkKey)
    {
        if (_recorder is null) return;
        _recorder.KeyUp(ahkKey);
        ShowRecordedTrigger();
        if (_recorder.IsComplete) StopRecording();
    }

    public ImportResult ImportFromScript(string scriptText, string defaultGroup)
    {
        var result = AhkImporter.Import(scriptText, defaultGroup);
        HotkeyItemViewModel? first = null;
        foreach (var hotkey in result.Hotkeys)
        {
            var item = HotkeyItemViewModel.FromHotkey(hotkey with { Id = Guid.NewGuid() });
            AddItem(item);
            first ??= item;
        }
        if (first is not null) SelectedItem = first;

        var message = $"Imported {result.Hotkeys.Count} hotkey(s). Stop the original script, then click Apply to avoid duplicate hotkeys.";
        if (result.SkippedLines.Count > 0)
        {
            message += $" {result.SkippedLines.Count} line(s) could not be imported: {string.Join(" | ", result.SkippedLines.Take(3))}";
        }
        SetStatus(message, result.SkippedLines.Count > 0 ? InfoBarSeverity.Warning : InfoBarSeverity.Success);
        return result;
    }

    private bool HasSelection() => SelectedItem is not null;

    private void AddItem(HotkeyItemViewModel item)
    {
        item.PropertyChanged += OnItemChanged;
        Items.Add(item);
    }

    private void OnItemChanged(object? sender, PropertyChangedEventArgs e)
    {
        if (e.PropertyName is nameof(HotkeyItemViewModel.Error) or nameof(HotkeyItemViewModel.Summary)) return;
        IsDirty = true;
        RefreshPreview();
    }

    private void RefreshPreview() => Preview = _services.Generator.Generate(Items.Select(i => i.ToHotkey())).Text;

    private void BeginRecording()
    {
        _recorder = new TriggerRecorder();
        // AutoHotkey intercepts keys before this window sees them, so pause our own script while recording.
        _resumeAfterRecording = _services.Runner.IsRunning(_scriptPath);
        if (_resumeAfterRecording) _services.Runner.Stop(_scriptPath);
        IsRecording = true;
    }

    private void StopRecording()
    {
        _recorder = null;
        IsRecording = false;
        if (!_resumeAfterRecording) return;
        _resumeAfterRecording = false;
        try { _services.Runner.Start(_scriptPath); }
        catch (AhkNotFoundException) { SetStatus("AutoHotkey v2 was not found. Set its path in Settings.", InfoBarSeverity.Error); }
    }

    private void ShowRecordedTrigger()
    {
        if (_recorder?.Trigger is { } trigger && SelectedItem is not null) SelectedItem.Trigger = trigger;
    }

    private void SetStatus(string message, InfoBarSeverity severity)
    {
        StatusMessage = message;
        StatusSeverity = severity;
        IsStatusOpen = true;
    }
}
```

```csharp
namespace F75Studio.App.ViewModels;

public sealed partial class ScriptsViewModel : ObservableObject
{
    private readonly AppServices _services;
    private readonly HotkeysViewModel _hotkeys;
    private readonly Action<object> _navigate;

    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(StartCommand), nameof(StopCommand), nameof(RestartCommand), nameof(SaveCommand),
        nameof(ImportIntoBuilderCommand), nameof(OpenInVsCodeCommand))]
    private ScriptItemViewModel? _selectedScript;

    [ObservableProperty] private TextDocument _document = new();
    [ObservableProperty, NotifyCanExecuteChangedFor(nameof(SaveCommand))] private bool _isEditorDirty;
    [ObservableProperty] private string? _statusMessage;
    [ObservableProperty] private bool _isStatusOpen;
    [ObservableProperty] private InfoBarSeverity _statusSeverity = InfoBarSeverity.Informational;

    public ScriptsViewModel(AppServices services, HotkeysViewModel hotkeys, Action<object> navigate)
    {
        _services = services;
        _hotkeys = hotkeys;
        _navigate = navigate;
        services.Runner.RunningChanged += (_, _) => Ui.Run(UpdateRunningState);
        Refresh();
    }

    public ObservableCollection<ScriptItemViewModel> Scripts { get; } = [];

    [RelayCommand]
    private void Refresh()
    {
        var selectedPath = SelectedScript?.Path;
        var generated = Path.GetFullPath(_services.Paths.GeneratedHotkeysScript);
        Scripts.Clear();
        foreach (var info in ScriptDiscovery.Discover(_services.Paths.ScriptSearchFolders))
        {
            Scripts.Add(new ScriptItemViewModel(
                info,
                isGenerated: string.Equals(info.Path, generated, StringComparison.OrdinalIgnoreCase),
                isRunning: _services.Runner.IsRunning(info.Path),
                runAtStartup: _services.Settings.StartupScripts.Contains(info.Path, StringComparer.OrdinalIgnoreCase),
                onRunAtStartupChanged: SetRunAtStartup));
        }
        SelectedScript = Scripts.FirstOrDefault(s => s.Path == selectedPath) ?? Scripts.FirstOrDefault();
    }

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void Start() => Guard(() =>
    {
        _services.Runner.Start(SelectedScript!.Path);
        SetStatus($"Started {SelectedScript.Name}.", InfoBarSeverity.Success);
    });

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void Stop() => Guard(() =>
    {
        _services.Runner.Stop(SelectedScript!.Path);
        SetStatus($"Stopped {SelectedScript.Name}.", InfoBarSeverity.Informational);
    });

    // AhkRunner.Start stops a running instance first, so Restart is Start.
    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void Restart() => Start();

    [RelayCommand(CanExecute = nameof(CanSave))]
    private void Save()
    {
        var script = SelectedScript!;
        ScriptFile.SaveWithBackup(script.Path, Document.Text);
        IsEditorDirty = false;
        if (_services.Runner.IsRunning(script.Path)) Guard(() => _services.Runner.Start(script.Path));
        SetStatus($"Saved {script.Name}. The previous version is kept as {script.Name}.bak.", InfoBarSeverity.Success);
    }

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void ImportIntoBuilder()
    {
        _hotkeys.ImportFromScript(Document.Text, Path.GetFileNameWithoutExtension(SelectedScript!.Name));
        _navigate(_hotkeys);
    }

    [RelayCommand(CanExecute = nameof(HasSelection))]
    private void OpenInVsCode()
    {
        try
        {
            Process.Start(new ProcessStartInfo("cmd.exe", $"/c code \"{SelectedScript!.Path}\"") { CreateNoWindow = true, UseShellExecute = false });
        }
        catch (Exception ex)
        {
            SetStatus($"Could not start VS Code: {ex.Message}", InfoBarSeverity.Error);
        }
    }

    partial void OnSelectedScriptChanged(ScriptItemViewModel? value)
    {
        var document = new TextDocument(value is null ? "" : File.ReadAllText(value.Path));
        document.TextChanged += (_, _) => IsEditorDirty = true;
        Document = document;
        IsEditorDirty = false;
    }

    private bool HasSelection() => SelectedScript is not null;

    private bool CanSave() => SelectedScript is { IsGenerated: false } && IsEditorDirty;

    private void SetRunAtStartup(ScriptItemViewModel item, bool enabled) =>
        _services.UpdateSettings(s => s with
        {
            StartupScripts = enabled
                ? s.StartupScripts.Append(item.Path).Distinct(StringComparer.OrdinalIgnoreCase).ToList()
                : s.StartupScripts.Where(p => !p.Equals(item.Path, StringComparison.OrdinalIgnoreCase)).ToList(),
        });

    private void UpdateRunningState()
    {
        foreach (var script in Scripts) script.IsRunning = _services.Runner.IsRunning(script.Path);
    }

    private void Guard(Action action)
    {
        try { action(); }
        catch (AhkNotFoundException) { SetStatus("AutoHotkey v2 was not found. Set its path in Settings.", InfoBarSeverity.Error); }
    }

    private void SetStatus(string message, InfoBarSeverity severity)
    {
        StatusMessage = message;
        StatusSeverity = severity;
        IsStatusOpen = true;
    }
}
```

```csharp
namespace F75Studio.Hotkeys;

/// <summary>Stops every running script and remembers which ones to start again.</summary>
public sealed class HotkeyPauser
{
    private readonly IAhkRunner _runner;
    private List<string> _paused = [];

    public HotkeyPauser(IAhkRunner runner) => _runner = runner;

    public bool IsPaused { get; private set; }

    public void Pause()
    {
        if (IsPaused) return;
        _paused = _runner.RunningScripts.ToList();
        _runner.StopAll();
        IsPaused = true;
    }

    public void Resume()
    {
        if (!IsPaused) return;
        foreach (var script in _paused.Where(File.Exists)) _runner.Start(script);
        _paused = [];
        IsPaused = false;
    }
}
```

```csharp
using System.Windows.Controls;
using CommunityToolkit.Mvvm.Input;
using F75Studio.Hotkeys;
using H.NotifyIcon;

namespace F75Studio.App.Services;

public sealed class TrayService : IDisposable
{
    private readonly AppServices _services;
    private readonly HotkeyPauser _pauser;
    private readonly Action _show;
    private readonly Action _exit;
    private readonly TaskbarIcon _icon;

    public TrayService(AppServices services, Action show, Action exit)
    {
        _services = services;
        _pauser = new HotkeyPauser(services.Runner);
        _show = show;
        _exit = exit;

        var menu = new ContextMenu();
        menu.Opened += (_, _) => BuildMenu(menu);
        BuildMenu(menu);

        _icon = new TaskbarIcon
        {
            ToolTipText = "F75 Studio",
            Icon = System.Drawing.Icon.ExtractAssociatedIcon(Environment.ProcessPath!),
            ContextMenu = menu,
            NoLeftClickDelay = true,
            LeftClickCommand = new RelayCommand(show),
        };
        _icon.ForceCreate(enablesEfficiencyMode: false);
    }

    public void Notify(string title, string message) => _icon.ShowNotification(title, message);

    public void Dispose() => _icon.Dispose();

    /// <summary>Rebuilt every time the menu opens so script states are current.</summary>
    private void BuildMenu(ContextMenu menu)
    {
        menu.Items.Clear();
        menu.Items.Add(Item("Open F75 Studio", _show));
        menu.Items.Add(new Separator());

        var scripts = new MenuItem { Header = "Scripts" };
        foreach (var script in ScriptDiscovery.Discover(_services.Paths.ScriptSearchFolders))
        {
            var running = _services.Runner.IsRunning(script.Path);
            var item = new MenuItem { Header = script.Name, IsCheckable = true, IsChecked = running };
            item.Click += (_, _) => Guard(() =>
            {
                if (running) _services.Runner.Stop(script.Path);
                else _services.Runner.Start(script.Path);
            });
            scripts.Items.Add(item);
        }
        if (scripts.Items.Count == 0) scripts.Items.Add(new MenuItem { Header = "No scripts found", IsEnabled = false });
        menu.Items.Add(scripts);

        menu.Items.Add(Item(_pauser.IsPaused ? "Resume hotkeys" : "Pause all hotkeys", TogglePause));
        menu.Items.Add(new Separator());
        menu.Items.Add(Item("Exit", _exit));
    }

    private void TogglePause() => Guard(() =>
    {
        if (_pauser.IsPaused) _pauser.Resume();
        else _pauser.Pause();
        _icon.ToolTipText = _pauser.IsPaused ? "F75 Studio (hotkeys paused)" : "F75 Studio";
    });

    private void Guard(Action action)
    {
        try { action(); }
        catch (AhkNotFoundException) { Notify("AutoHotkey not found", "Set the AutoHotkey v2 path in F75 Studio Settings."); }
    }

    private static MenuItem Item(string header, Action action)
    {
        var item = new MenuItem { Header = header };
        item.Click += (_, _) => action();
        return item;
    }
}
```
